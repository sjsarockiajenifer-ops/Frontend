Task 8: Orders Table & Conditional Record Deletions (DELETE)

task8.sql
MySQL 8.0 • college_db 
use college_db;

create table orders (
    order_id int primary key auto_increment,
    customer_name varchar(100),
    product_name varchar(100),
    quantity int,
    price decimal(10,2),
    order_date date
);
